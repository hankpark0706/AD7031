\problem{25}{Problem 1. Mean and covariance of a random vector}

We have three independent identically distributed (i.i.d.) return data points:

\begin{center}
\begin{tabular}{c|ccc}
           & Tech   & Energy & Bond   \\ \hline
 $\xi^1$   & $0.00$ & $0.04$ & $0.03$ \\
 $\xi^2$   & $0.06$ & $0.10$ & $0.03$ \\
 $\xi^3$   & $0.12$ & $0.07$ & $0.03$ \\
\end{tabular}
\end{center}

Note that we use $\widehat{\mathbb{E}}[\cdot]$, $\widehat{\mathrm{Var}}[\cdot]$ and $\widehat{\mathrm{Cov}}[\cdot]$ for empirical
estimates, to distinguish them from their true counterparts $\mathbb{E}[\cdot]$, $\mathrm{Var}[\cdot]$ and $\mathrm{Cov}[\cdot]$.

\begin{parts}
\item \textbf{(8 pts)} \textbf{By hand}, compute the empirical mean $\widehat{\mu} = \widehat{\mathbb{E}}[\tilde\xi]$ and the
  empirical covariance matrix $\widehat{\Sigma} = \widehat{\mathrm{Cov}}[\tilde\xi]$, and state the dimensions of $\widehat{\mu}$ and $\widehat{\Sigma}$.

\item \textbf{(5 pts)} We used only three data points. In general, with $N$ i.i.d.\ samples of $\tilde\xi$, the empirical mean and
  covariance matrix can be written $\widehat{\mu}_N$ and $\widehat{\Sigma}_N$. Consider the claim
  \[
    \lim_{N \to \infty} \widehat{\mu}_N = \mathbb{E}[\tilde\xi] = \mu, \qquad
    \lim_{N \to \infty} \widehat{\Sigma}_N = \mathrm{Cov}[\tilde\xi] = \Sigma .
  \]
  What does it mean? Explain in plain Korean. Is it true? If so, why?

\item \textbf{(7 pts)} Suppose your optimal portfolio is $x^{\star} = (x_1^{\star}, x_2^{\star}, x_3^{\star})^{\top}$.
  From the table, compute, in terms of $x^{\star}$,
  \begin{itemize}[leftmargin=2.2em, itemsep=2pt, topsep=3pt]
    \item the empirical mean $\widehat{\mathbb{E}}[\tilde\xi^{\top}x^{\star}]$;
    \item the empirical variance $\widehat{\mathrm{Var}}[\tilde\xi^{\top}x^{\star}]$.
  \end{itemize}

\item \textbf{(5 pts)} Show that
  \begin{itemize}[leftmargin=2.2em, itemsep=2pt, topsep=3pt]
    \item $\widehat{\mathbb{E}}[\tilde\xi^{\top}x^{\star}] = \widehat{\mu}^{\top}x^{\star}$;
    \item $\widehat{\mathrm{Var}}[\tilde\xi^{\top}x^{\star}] = x^{\star\top}\widehat{\Sigma}\, x^{\star}$.
  \end{itemize}
  \hint{we discussed this in class.}
\end{parts}

\problem{25}{Problem 2. Linear program duality}

Consider the following primal linear program and its dual, in the canonical form of the lecture note:
\[
\text{(P)} \quad
\begin{aligned}
  \min_{x \in \mathbb{R}^n} \quad & c^{\top}x \\
  \text{s.t.} \quad & Ax \le b
\end{aligned}
\qquad\qquad
\text{(D)} \quad
\begin{aligned}
  \max_{\theta \in \mathbb{R}^m} \quad & -b^{\top}\theta \\
  \text{s.t.} \quad & c + A^{\top}\theta = 0 \\
                    & \theta \ge 0
\end{aligned}
\]

\begin{parts}
\item \textbf{(8 pts)} Derive the dual problem (D) from the primal problem (P). \textbf{Show all the steps clearly.}

\item \textbf{(7 pts)} Consider the primal problem
\[
\begin{aligned}
  \min_{x \in \mathbb{R}^2} \quad & -4x_1 - 5x_2 \\
  \text{s.t.} \quad & 3x_1 + 2x_2 \le 60 \\
                    & x_1 + 3x_2 \le 48 \\
                    & x_1 \ge 3 \\
                    & x_1,\ x_2 \ge 0 .
\end{aligned}
\]
  \begin{itemize}[leftmargin=2.2em, itemsep=2pt, topsep=3pt]
    \item Give $c$, $A$, $b$.
    \item Write the dual problem (D) numerically.
  \end{itemize}
  \hint{follow the same steps as in (a), this time with these numbers.}

\item \textbf{(7 pts)} Derive the primal problem (P) from the dual problem (D). \textbf{Show all the steps clearly.}
  \hint{nothing really changes from the derivation in (a).}

\item \textbf{(3 pts)} Let $x^{\star}$ and $\theta^{\star}$ be optimal solutions of the primal problem (P) and the dual problem (D),
  respectively. What can you say about $c^{\top}x^{\star}$ and $b^{\top}\theta^{\star}$?
\end{parts}

\problem{25}{Problem 3. Conic program duality}

Consider
\[
\begin{aligned}
  \min_{x \in \mathbb{R}^n} \quad & x^{\top}\Sigma x \\
  \text{s.t.} \quad & \mu^{\top}x \ge r \\
                    & e^{\top}x \le 1 \\
                    & x \ge 0
\end{aligned}
\]
where $e = [1, 1, \dots, 1]^{\top}$ is the all-ones vector and $r$ is the target return.

\begin{parts}
\item \textbf{(12 pts)} Recall the second-order cone
  \[
    \mathrm{SOC}_Q^{n+1} = \big\{ (x, t) \in \mathbb{R}^n \times \mathbb{R} \;:\; x^{\top}Q\,x \le t^2,\ t \ge 0 \big\},
    \qquad Q \succ 0 .
  \]
  Write the model in the conic form $\min c^{\top}z$ s.t.\ $Az \le b$, $z \in C$.
  \begin{itemize}[leftmargin=2.2em, itemsep=2pt, topsep=3pt]
    \item Give $z$ (the decision variable), $c$, $A$, $b$.
    \item Write $C$.
  \end{itemize}
  \hint[1]{first, use the epigraph reformulation with a new decision variable $t$, then define
    $z = [x,\ t]^{\top}$.}
  \hint[2]{minimizing $x^{\top}\Sigma x$ and minimizing $\sqrt{x^{\top}\Sigma x}$ give the same $x$.}

\item \textbf{(13 pts)} With $\Sigma = I$, the dual problem of the second-order cone program in (a) is
  \[
  \begin{aligned}
    \max \quad & r\,\theta_1 - \theta_2 \\
    \text{s.t.} \quad & \theta_1,\ \theta_2 \ge 0, \quad \theta_3 \in \mathbb{R}^n_+ \\
                      & \|\theta_2 e - \theta_1 \mu - \theta_3\|_2 \;\le\; 1 .
  \end{aligned}
  \]
  Derive it. \textbf{Show all the steps clearly.}
\end{parts}
