\beat{1}{Random variables, events, conditioning}
\begin{board}
back to week 2's Section 2 table: the OR column had \textbf{uncertain data} $\xi$. today, the vocabulary ---
written in this course's notation from the first line --- $x$ is already taken by the
\textbf{decision}, so the uncertain data is $\tilde\xi$ throughout.
\vspace{10pt}
\vspace{10pt}

\bul \textbf{random experiment:} its outcome is not known in advance
\vspace{3pt}

\bul \textbf{sample space} $\Omega$ = the set of all possible outcomes
\vspace{6pt}
\vspace{10pt}

an \textbf{event} is a subset of the sample space, \ $A \subseteq \Omega$; \ $\mathbb{P}(A)$ is the probability it occurs.
\vspace{6pt}

\bul \ $0 \le \mathbb{P}(A) \le 1$, \qquad $\mathbb{P}(\Omega) = 1$
\vspace{3pt}

\bul \ $\mathbb{P}(A \cup B) = \mathbb{P}(A) + \mathbb{P}(B) - \mathbb{P}(A \cap B)$ \qquad --- subtract the overlap, or it is counted twice
\vspace{10pt}
\vspace{12pt}

\textbf{Def} (\emph{random variable}) \quad a function that assigns a real number to each outcome: \quad $\tilde\xi : \Omega \to \mathbb{R}$
\vspace{3pt}

\quad the tilde marks the \emph{random} object; \ $\xi$ without it is a \textbf{realization} --- one value it took
\vspace{5pt}

\quad toss a coin twice: \ $\Omega = \{HH, HT, TH, TT\}$, \ each outcome has probability $\tfrac14$
\vspace{3pt}

\quad $\tilde\xi$ = number of heads: \quad $\tilde\xi(HH) = 2, \quad \tilde\xi(HT) = 1, \quad \tilde\xi(TH) = 1, \quad \tilde\xi(TT) = 0$
\vspace{6pt}

\begin{center}

\vspace{6pt}
\vspace{6pt}

\bul \textbf{discrete} $\tilde\xi$: countably many values --- number of defective parts, number of arrivals, demand
\vspace{3pt}

\bul \textbf{continuous} $\tilde\xi$: values fill an interval --- length, time, temperature, a return
\vspace{6pt}

\bul \textbf{support} $\Xi$ = the set of values $\tilde\xi$ actually takes \ (the smallest closed set with $\mathbb{P}(\tilde\xi \in \Xi) = 1$)
\vspace{3pt}

\quad --- $\Omega$ is where the \emph{experiment} lives, $\Xi \subseteq \mathbb{R}^k$ is where the \emph{numbers} live
\vspace{14pt}
\vspace{14pt}

\textbf{Def} (\emph{conditional probability}) \quad
$\mathbb{P}(A \mid B) = \dfrac{\mathbb{P}(A \cap B)}{\mathbb{P}(B)}$, \qquad defined when $\mathbb{P}(B) > 0$
\vspace{3pt}

\quad --- $B$ becomes the new sample space; divide by $\mathbb{P}(B)$ so the weights add to $1$ again
\vspace{8pt}

\quad two coins: \quad $A = \{\text{both flips are heads}\} = \{HH\}$, \qquad $B = \{\text{first flip is head}\} = \{HH, HT\}$
\vspace{3pt}

\quad $\mathbb{P}(A) = \tfrac14$, \quad $\mathbb{P}(B) = \tfrac12$, \quad $A \cap B = \{HH\}$:
\qquad $\mathbb{P}(A \mid B) = \dfrac{1/4}{1/2} = \tfrac12$
\vspace{3pt}

\quad --- the first head doubles it, $\tfrac14 \to \tfrac12$
\vspace{10pt}

\textbf{Def} (\emph{independent events}) \quad $A$ and $B$ are independent if \quad $\mathbb{P}(A \cap B) = \mathbb{P}(A)\,\mathbb{P}(B)$
\vspace{3pt}

\quad then \ $\mathbb{P}(A \mid B) = \dfrac{\mathbb{P}(A)\,\mathbb{P}(B)}{\mathbb{P}(B)} = \mathbb{P}(A)$ \quad --- learning $B$ tells you nothing about $A$
\vspace{8pt}

\quad same coins, \ same $B$. \quad now $A = \{\text{exactly one head}\} = \{HT, TH\}$:
\vspace{3pt}

\quad $\mathbb{P}(A \cap B) = \mathbb{P}(\{HT\}) = \tfrac14 = \tfrac12 \cdot \tfrac12 = \mathbb{P}(A)\,\mathbb{P}(B)$
\qquad --- \textbf{independent}
\vspace{3pt}

\quad so $\mathbb{P}(A \mid B) = \tfrac12 = \mathbb{P}(A)$: \ the first flip says nothing about whether there is exactly one head.
      \ {\color{punchink}with $A = $ both heads, above, it said plenty} --- independence is a property of the \emph{pair}.
\vspace{10pt}

\textbf{Thm} (\emph{law of total probability}) \quad let $A_1, \dots, A_I$ \textbf{partition} $\Omega$:
\ $A_i \cap A_j = \emptyset$ for $i \ne j$, \ and \ $\bigcup_{i} A_i = \Omega$. \ then for any event $B$,
\[
\mathbb{P}(B) \;=\; \mathbb{P}(B \cap A_1) + \cdots + \mathbb{P}(B \cap A_I)
     \;=\; \mathbb{P}(B \mid A_1)\,\mathbb{P}(A_1) + \cdots + \mathbb{P}(B \mid A_I)\,\mathbb{P}(A_I)
\]

\quad --- split the world into cases, answer each case, weight by how likely the case is
\vspace{10pt}

\quad two coins again: \ $A_1 = B$ (first flip head), \ $A_2 = B^{c}$ (first flip tail) --- a partition, each $\tfrac12$
\vspace{4pt}

\quad the event of interest: \ \emph{at least one head} \qquad $\mathbb{P}(\text{at least one head} \mid A_1) = 1$ \ (a head already landed), \qquad
      $\mathbb{P}(\text{at least one head} \mid A_2) = \tfrac12$ \ (needs the second)
\vspace{4pt}

\quad $\mathbb{P}(\text{at least one head}) = 1 \cdot \tfrac12 \;+\; \tfrac12 \cdot \tfrac12 \;=\; \tfrac34$ \qquad
      --- the same $\tfrac34$ Section 2 gives from the scenario list
\end{board}
\takeaway{conditioning is how a \emph{two-stage} decision gets written down: act, watch which $A_i$ occurred, act again. weeks 10--12 run on this.}

\beat{2}{Discrete random variables}
\begin{board}
a discrete $\tilde\xi$ is a \textbf{list}: finitely many values, each carrying a weight.
\vspace{10pt}

\textbf{Def} (\emph{scenarios}) \quad $\tilde\xi$ takes the values \ $\xi^1, \dots, \xi^S$ --- the \textbf{scenarios} --- with
\vspace{4pt}

\quad $p_s = \mathbb{P}\big(\tilde\xi = \xi^s\big)$, \qquad
(1) \ $p_s \ge 0$ \qquad (2) \ $\sum_{s \in [S]} p_s = 1$, \qquad $[S] = \{1, \dots, S\}$
\vspace{4pt}

\quad $\mathbb{P}(\tilde\xi \in A) = \sum_{s \,:\, \xi^s \in A} p_s$ \qquad --- add the weights of the scenarios landing in $A$
\vspace{6pt}

\quad {\color{punchink}the scenario index goes \emph{up}:} \ $\xi^s$ is scenario $s$; \ a \emph{subscript} $\xi_j$ is
      component $j$ of a vector (Section 7). two different jobs, two different slots.
\vspace{6pt}

\quad older books call this list the \textbf{probability mass function} and write it as a function $f$.
      same content --- but a table, and a table is the only thing a solver can be handed.
\vspace{10pt}

\quad two coins: \quad $S = 3$, \qquad $\xi^1 = 0$, \ $\xi^2 = 1$, \ $\xi^3 = 2$, \qquad
      $p = \big(\tfrac14, \tfrac12, \tfrac14\big)$
\vspace{3pt}

\quad $\mathbb{P}(\tilde\xi \ge 1) = p_2 + p_3 = \tfrac34$ \qquad --- the $\tfrac34$ Section 1 reached by conditioning
\vspace{12pt}

\textbf{Def} (\emph{mean}) \quad $\mu = \mathbb{E}[\tilde\xi] = \sum_{s \in [S]} p_s\, \xi^s$
\vspace{6pt}

\quad function of $\tilde\xi$: \quad $\mathbb{E}\big[h(\tilde\xi)\big] = \sum_{s \in [S]} p_s\, h(\xi^s)$
\qquad {\color{punchink}\small $\leftarrow$ law of the unconscious statistician}
\vspace{3pt}

\quad --- keep the weights $p_s$, change only what they weight. no new distribution needed
\vspace{12pt}

\textbf{Linearity of expectation} \quad $\mathbb{E}$ is a weighted sum, so constants pull out and sums split:
\vspace{4pt}

\quad $\mathbb{E}[c] = c$, \qquad
      $\mathbb{E}[a\tilde\xi + b] = a\,\mathbb{E}[\tilde\xi] + b$, \qquad
      $\mathbb{E}\big[g(\tilde\xi) + h(\tilde\xi)\big] = \mathbb{E}\big[g(\tilde\xi)\big] + \mathbb{E}\big[h(\tilde\xi)\big]$
\vspace{3pt}

\quad --- each one is $\sum_s p_s(\cdots)$ pulled apart; there is nothing deeper in it
\vspace{12pt}

\textbf{Def} (\emph{variance}) \quad
$\sigma^2 = \mathrm{Var}(\tilde\xi) = \mathbb{E}\big[(\tilde\xi - \mu)^2\big] = \sum_{s \in [S]} p_s\, \big(\xi^s - \mu\big)^2$,
\qquad $\sigma = \sqrt{\mathrm{Var}(\tilde\xi)}$
\vspace{4pt}

\quad the \textbf{shortcut} --- expand the square, then apply linearity term by term:
\[
\begin{aligned}
\mathrm{Var}(\tilde\xi)
  &= \mathbb{E}\big[(\tilde\xi - \mu)^2\big]
   = \mathbb{E}\big[\tilde\xi^2 - 2\mu\,\tilde\xi + \mu^2\big] \\[2pt]
  &= \mathbb{E}\big[\tilde\xi^2\big] - 2\mu\,\mathbb{E}[\tilde\xi] + \mu^2
   = \mathbb{E}\big[\tilde\xi^2\big] - 2\mu^2 + \mu^2
   = \mathbb{E}\big[\tilde\xi^2\big] - \mu^2
\end{aligned}
\]

\quad {\color{punchink}$\mu$ is a number, not random} --- that is what lets it come out of the $\mathbb{E}$
\vspace{6pt}

\quad $\mathrm{Var}(a\tilde\xi + b) = a^2\,\mathrm{Var}(\tilde\xi)$
\qquad --- shifting by $b$ moves the centre, not the spread
\vspace{12pt}

\textbf{Example 1} \quad demand $\tilde\xi$ for a spare part next month:
\vspace{5pt}

\begin{center}
\begin{tabular}{@{}l|ccc@{}}
 $\xi^s$ & 4    & 8   & 12   \\ \hline
 $p_s$   & 0.25 & 0.5 & 0.25 \\
\end{tabular}
\qquad\qquad
\begin{tikzpicture}[scale=0.95, font=\scriptsize, baseline=-6pt]
  \draw[dashed, red!75!black, thick] (1.6,0) -- (1.6,1.4) node[above] {$\mathbb{E}[\tilde\xi] = 8$};
  \draw[->] (-0.3,0) -- (3.6,0) node[right] {$\xi$};
  \draw[fill=blue!15, draw=blue!65!black] (0.35,0) rectangle (0.85,0.5);
  \draw[fill=blue!15, draw=blue!65!black] (1.35,0) rectangle (1.85,1.0);
  \draw[fill=blue!15, draw=blue!65!black] (2.35,0) rectangle (2.85,0.5);
  \node[below] at (0.6,0) {4}; \node[below] at (1.6,0) {8}; \node[below] at (2.6,0) {12};

\end{tikzpicture}
\end{center}
\vspace{5pt}

\quad $\mathbb{E}[\tilde\xi] = 0.25 \cdot 4 + 0.5 \cdot 8 + 0.25 \cdot 12 = 8$
\vspace{3pt}

\quad $\mathrm{Var}(\tilde\xi) = 0.25 \cdot (4 - 8)^2 + 0.5 \cdot (8 - 8)^2 + 0.25 \cdot (12 - 8)^2 = 8$,
\qquad $\sigma = 2\sqrt{2} \approx 2.83$
\vspace{14pt}
\end{board}

\beat{3}{Continuous random variables}
\begin{board}
\textbf{Def} (\emph{probability density function, PDF}) \quad $f : \mathbb{R} \to \mathbb{R}_+$ with
\vspace{3pt}

\quad (1) \ $p(\xi) \ge 0$ \qquad (2) \ $\int_{-\infty}^{\infty} p(\xi)\, d\xi = 1$ \qquad
(3) \ $\mathbb{P}(a \le \tilde\xi \le b) = \int_a^b p(\xi)\, d\xi$ \ = area under $p$ from $a$ to $b$
\vspace{3pt}

\quad {\color{punchink}$p(\xi)$ is \emph{not} a probability:} \ $\mathbb{P}(\tilde\xi = \xi) = 0$ for every single $\xi$, \ and $p(\xi) > 1$ is allowed
\vspace{8pt}

\textbf{Def} (\emph{mean, variance}) \quad same as discrete --- the sums become integrals:
\vspace{3pt}

\quad $\mu = \mathbb{E}[\tilde\xi] = \int \xi\, p(\xi)\, d\xi$, \qquad
$\sigma^2 = \mathrm{Var}(\tilde\xi) = \int (\xi - \mu)^2 p(\xi)\, d\xi = \mathbb{E}[\tilde\xi^2] - \mu^2$, \qquad
$\mathbb{E}[h(\tilde\xi)] = \int h(\xi)\, p(\xi)\, d\xi$
\vspace{3pt}

\quad the shortcut $\mathrm{Var}(\tilde\xi) = \mathbb{E}[\tilde\xi^2] - \mu^2$ carries over unchanged: \ Section 2 derived it
      from linearity alone, and linearity holds here for the same reason --- the integral splits and constants come out
\vspace{10pt}

\textbf{Example 2} \quad $\tilde\xi \sim \mathrm{Uniform}[a,b]$ \ ---
$p(\xi) = \dfrac{1}{b-a}$ \ on \ $[a,b]$, \ $0$ outside
\vspace{4pt}

\begin{center}
\begin{tikzpicture}[xscale=0.30, yscale=7.5, font=\scriptsize]
  \draw[dashed, gray!70] (0,0.125) -- (4,0.125);
  \draw[->] (-0.5,0) -- (15.5,0) node[right] {$\xi$};
  \draw[->] (0,-0.012) -- (0,0.195) node[left] {$p(\xi)$};
  \draw[fill=blue!15, draw=blue!65!black, thick] (4,0) rectangle (12,0.125);
  \node[below] at (4,0) {$a = 4$};
  \node[below] at (12,0) {$b = 12$};
  \node[left] at (0,0.125) {$\tfrac{1}{b-a}$};
  \draw[dashed, red!75!black, thick] (8,0) -- (8,0.165) node[above] {$\mathbb{E}[\tilde\xi] = 8$};
  \node[below] at (8,-0.022) {$8$};
\end{tikzpicture}
\end{center}
\vspace{4pt}

\quad mean:
\[
\mathbb{E}[\tilde\xi] \;=\; \int_a^b \xi \, \frac{1}{b-a} \, d\xi
 \;=\; \frac{1}{b-a}\left[\frac{x^2}{2}\right]_a^b
 \;=\; \frac{b^2 - a^2}{2(b-a)} \;=\; \frac{a+b}{2}
\]

\quad second moment --- $h(x) = x^2$:
\[
\mathbb{E}[\tilde\xi^2] \;=\; \int_a^b \xi^2 \, \frac{1}{b-a} \, d\xi
 \;=\; \frac{1}{b-a}\left[\frac{x^3}{3}\right]_a^b
 \;=\; \frac{b^3 - a^3}{3(b-a)} \;=\; \frac{a^2 + ab + b^2}{3}
\]

\quad variance --- subtract, no new integral:
\[
\mathrm{Var}(\tilde\xi) \;=\; \mathbb{E}[\tilde\xi^2] - \mathbb{E}[\tilde\xi]^2
 \;=\; \frac{a^2 + ab + b^2}{3} - \frac{(a+b)^2}{4}
 \;=\; \frac{(b-a)^2}{12}
\]

\quad with $a = 4$, $b = 12$: \qquad
$\mathbb{E}[\tilde\xi] = 8$, \qquad
$\mathrm{Var}(\tilde\xi) = \dfrac{64}{12} = \dfrac{16}{3}$, \qquad
$\sigma = \dfrac{4}{\sqrt{3}} \approx 2.31$
\vspace{12pt}

\textbf{Normal distribution} \quad $\tilde\xi \sim \mathcal{N}(\mu, \sigma^2)$:
\qquad $p(\xi) = \dfrac{1}{\sigma\sqrt{2\pi}}\, \exp\!\Big(\!-\dfrac{(\xi-\mu)^2}{2\sigma^2}\Big)$, \ $\xi \in \mathbb{R}$
\vspace{4pt}

\quad $\mathbb{E}[\tilde\xi] = \mu$, \qquad $\mathrm{Var}(\tilde\xi) = \sigma^2$
\vspace{3pt}

\quad $\mathbb{P}(\mu - \sigma \le \tilde\xi \le \mu + \sigma) \approx 0.68$, \qquad
$\mathbb{P}(\mu - 2\sigma \le \tilde\xi \le \mu + 2\sigma) \approx 0.95$
\vspace{4pt}

\begin{center}

\beat{4}{Expectation, in one formula}
\begin{board}
Sections 2 and 3 defined the mean twice --- once a sum, once an integral. one definition covers both.
\vspace{10pt}

\textbf{Def} (\emph{expectation}) \quad for $\tilde\xi \sim \mathbb{P}$,
\[
\mathbb{E}_{\mathbb{P}}\big[\tilde\xi\big] \;=\; \int_{\Omega} \xi \;\, \mathbb{P}(d\xi)
\]

\quad read $\mathbb{P}(d\xi)$ as \emph{the probability sitting on a small piece} $d\xi$ --- the weight, whichever form it takes:
\vspace{6pt}

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}l|l|l@{}}
                & the weight $\mathbb{P}(d\xi)$ is        & and $\mathbb{E}_{\mathbb{P}}[\tilde\xi]$ reads \\ \hline
discrete        & $p_s$, \ one per scenario $\xi^s$       & $\sum_{s \in [S]} p_s\, \xi^s$ \\
continuous      & $p(\xi)\, d\xi$                         & $\int_{\Xi} \xi\, p(\xi)\, d\xi$ \\
\end{tabular}
\end{center}
\vspace{10pt}

\textbf{Def} (\emph{expectation of a function}) \quad for $f : \mathbb{R}^k \to \mathbb{R}$ and a random vector $\tilde\xi$,
\[
\mathbb{E}_{\mathbb{P}}\big[f(\tilde\xi)\big] \;=\; \int_{\Omega} f(\xi) \;\, \mathbb{P}(d\xi)
\qquad\text{---}\qquad
\sum_{s \in [S]} p_s\, f(\xi^s)
\quad\text{or}\quad
\int_{\Xi} f(\xi)\, p(\xi)\, d\xi
\]

\quad {\color{punchink}this is the one the course actually needs:} \ $f$ will be a \textbf{cost}, and $\mathbb{E}[f(x, \tilde\xi)]$ the objective.
\vspace{10pt}

\quad variance is an expectation too, so it inherits the same two readings:
\vspace{4pt}

\quad $\mathrm{Var}(\tilde\xi) = \mathbb{E}\big[(\tilde\xi - \mathbb{E}[\tilde\xi])^2\big]$, \qquad
      $\sigma(\tilde\xi) = \sqrt{\mathrm{Var}(\tilde\xi)}$, \qquad
      $\mathrm{Cov}(\tilde\xi_1, \tilde\xi_2) = \mathbb{E}\big[(\tilde\xi_1 - \mathbb{E}[\tilde\xi_1])(\tilde\xi_2 - \mathbb{E}[\tilde\xi_2])\big]$
\end{board}
\takeaway{one definition, two computations. write $\mathbb{E}$ once and stop asking whether the data is discrete or continuous.}

%% ------------------------------------------------------------------ 6
\beat{5}{Two random variables}
\begin{board}
\textbf{Def} (\emph{joint PMF}) \quad
$p(\xi_1, \xi_2) = \mathbb{P}(\tilde\xi_1 = \xi_1, \ \tilde\xi_2 = \xi_2)$, \qquad $p \ge 0$, \qquad $\sum_x \sum_y p(\xi_1, \xi_2) = 1$
\vspace{6pt}

\textbf{Def} (\emph{marginal PMF}) \quad
$p_1(\xi_1) = \sum_y p(\xi_1, \xi_2)$, \qquad $p_2(\xi_2) = \sum_x p(\xi_1, \xi_2)$
\vspace{2pt}

\quad --- row sums and column sums, written in the \emph{margins} of the table
\vspace{10pt}

\textbf{Example 3} \quad two assets, returns tomorrow: \ $\tilde\xi_1$, $\tilde\xi_2$ each either down ($-0.10$) or up ($+0.20$)
\vspace{6pt}

\begin{center}
\begin{tabular}{@{}c|cc|c@{}}
 $p(\xi_1, \xi_2)$        & $\xi_2 = -0.10$ & $\xi_2 = 0.20$ & $p_1(\xi_1)$ \\ \hline
 $\xi_1 = -0.10$           & $0.3$ & $0.2$ & $0.5$ \\
 $\xi_1 = \phantom{-}0.20$ & $0.2$ & $0.3$ & $0.5$ \\ \hline
 $p_2(\xi_2)$              & $0.5$ & $0.5$ & $1$ \\
\end{tabular}
\end{center}
\vspace{4pt}

\quad each asset alone: down or up w.p.\ $\tfrac12$. \quad together: same direction w.p.\ $0.3 + 0.3 = 0.6$
\vspace{10pt}

\textbf{Def} (\emph{joint PDF}) \quad
$p : \mathbb{R}^2 \to \mathbb{R}_+$, \qquad $\iint p(\xi_1, \xi_2)\, d\xi_1\, d\xi_2 = 1$, \qquad
$\mathbb{P}\big((\tilde\xi_1, \tilde\xi_2) \in R\big) = \iint_R p(\xi_1, \xi_2)\, d\xi_1\, d\xi_2$
\vspace{2pt}

\quad --- the \textbf{volume} under $p$ above the region $R$
\vspace{3pt}

\quad marginal PDF: \ $p_1(\xi_1) = \int p(\xi_1, \xi_2)\, d\xi_2$ \quad --- the sum becomes an integral
\vspace{10pt}

\textbf{Example 4} \quad $(\tilde\xi_1, \tilde\xi_2)$ uniform on the unit square: \quad $p(\xi_1, \xi_2) = 1$ \ on $[0,1]^2$, \ $0$ outside
\vspace{4pt}

\begin{center}
\begin{tikzpicture}[scale=1.55, font=\scriptsize, baseline=(b)]
  \coordinate (b) at (0,0.5);
  \draw[->] (-0.12,0) -- (1.35,0) node[right] {$\xi_1$};
  \draw[->] (0,-0.12) -- (0,1.3) node[left] {$\xi_2$};
  \draw[fill=blue!6, draw=blue!65!black, thick] (0,0) rectangle (1,1);

  \node[below] at (1,0) {$1$}; \node[left] at (0,1) {$1$};
  \node[anchor=west, font=\tiny, red!75!black] at (0.52,0.58) {$\xi_1 + \xi_2 = 1$};
\end{tikzpicture}
\qquad\quad
\begin{minipage}{0.62\textwidth}
$\mathbb{P}(\tilde\xi_1 + \tilde\xi_2 \le 1) = \iint_R 1 \; d\xi_1\, d\xi_2$ = area of the triangle $R$ $= \tfrac12$
\vspace{6pt}

marginal: \ $p_1(\xi_1) = \int_0^1 1 \; d\xi_2 = 1$ on $[0,1]$ --- $\tilde\xi_1$ alone is Uniform$[0,1]$
\end{minipage}
\end{center}
\vspace{10pt}

\textbf{Def} (\emph{independence}) \quad $\tilde\xi_1$, $\tilde\xi_2$ are independent if \quad
$p(\xi_1, \xi_2) = p_1(\xi_1)\, p_2(\xi_2) \quad \forall\, \xi_1, \xi_2$ \qquad (PMF or PDF)
\vspace{3pt}

\quad Example 3: \ $p(-0.10, -0.10) = 0.3 \ne 0.5 \cdot 0.5$ \ --- \textbf{not} independent
\vspace{2pt}

\quad Example 4: \ $p(\xi_1, \xi_2) = 1 = 1 \cdot 1$ \ --- independent
\vspace{10pt}
\end{board}
%% ------------------------------------------------------------------ 7
\beat{6}{Covariance and correlation}
\begin{board}
\textbf{Def} (\emph{covariance, correlation})
\vspace{3pt}

\quad $\mathrm{Cov}(\tilde\xi_1, \tilde\xi_2) = \mathbb{E}\big[(\tilde\xi_1 - \mu_1)(\tilde\xi_2 - \mu_2)\big] = \mathbb{E}[\tilde\xi_1 \tilde\xi_2] - \mu_1 \mu_2$,
\qquad $\mathrm{Cov}(\tilde\xi_1, \tilde\xi_1) = \mathrm{Var}(\tilde\xi_1)$
\vspace{3pt}

\quad $\rho_{12} = \dfrac{\mathrm{Cov}(\tilde\xi_1, \tilde\xi_2)}{\sigma_1\, \sigma_2}$, \qquad $-1 \le \rho_{12} \le 1$
\vspace{3pt}

\quad $\rho > 0$: move together, \quad $\rho < 0$: move opposite, \quad independent $\Rightarrow$ $\rho = 0$
\vspace{4pt}

\begin{center}

\end{center}
\vspace{8pt}

\textbf{Example 3, continued} \quad (use the margins for $\mu$ and $\sigma$, all four cells for the covariance)
\vspace{4pt}

\quad $\mu_1 = 0.5 \cdot (-0.10) + 0.5 \cdot 0.20 = 0.05$, \qquad $\mu_2 = 0.05$
\vspace{3pt}

\quad $\mathrm{Var}(\tilde\xi_1) = 0.5 \cdot (-0.15)^2 + 0.5 \cdot (0.15)^2 = 0.0225$, \qquad $\sigma_1 = \sigma_2 = 0.15$
\[
\begin{aligned}
\mathrm{Cov}(\tilde\xi_1, \tilde\xi_2)
 &= 0.3\,(-0.15)(-0.15) + 0.2\,(-0.15)(0.15) + 0.2\,(0.15)(-0.15) + 0.3\,(0.15)(0.15) \\
 &= (0.3 - 0.2 - 0.2 + 0.3) \cdot 0.0225 \;=\; 0.0045
\end{aligned}
\]

\quad $\rho_{12} = \dfrac{0.0045}{0.15 \cdot 0.15} = 0.2$ \quad --- the two assets tend to move together
\vspace{14pt}
\end{board}
%% ------------------------------------------------------------------ 8
\beat{7}{Linear combinations}
\begin{board}
\quad $\mathbb{E}[a\tilde\xi_1 + b\tilde\xi_2] = a\,\mu_1 + b\,\mu_2$ \qquad --- always, no independence needed
\vspace{3pt}

\quad $\mathrm{Var}(a\tilde\xi_1 + b\tilde\xi_2) = a^2\,\mathrm{Var}(\tilde\xi_1) + b^2\,\mathrm{Var}(\tilde\xi_2) + 2ab\,\mathrm{Cov}(\tilde\xi_1, \tilde\xi_2)$
\vspace{8pt}

\quad $n$ random variables $\tilde\xi = (\tilde\xi_1, \dots, \tilde\xi_n)$:
\vspace{3pt}

\qquad mean vector \ $\mu = \big(\mathbb{E}[\tilde\xi_1], \dots, \mathbb{E}[\tilde\xi_n]\big)^{\top} \in \mathbb{R}^n$
\vspace{3pt}

\qquad covariance matrix \ $\Sigma \in \mathbb{R}^{n \times n}$, \ $\Sigma_{ij} = \mathrm{Cov}(\tilde\xi_i, \tilde\xi_j)$;
\ diagonal $\Sigma_{ii} = \mathrm{Var}(\tilde\xi_i)$; \ symmetric; \ always $\Sigma \succeq 0$
\vspace{4pt}

\quad for fixed $c \in \mathbb{R}^n$: \qquad
$\mathbb{E}[c^{\top}\tilde\xi] = c^{\top}\mu$, \qquad $\mathrm{Var}(c^{\top}\tilde\xi) = c^{\top}\Sigma c$
\vspace{4pt}

\quad $n = 2$, \ $c = (a, b)$: \quad
$c^{\top}\Sigma c
 = \begin{pmatrix} a & b \end{pmatrix}
   \begin{pmatrix} \mathrm{Var}(\tilde\xi_1) & \mathrm{Cov}(\tilde\xi_1, \tilde\xi_2) \\ \mathrm{Cov}(\tilde\xi_1, \tilde\xi_2) & \mathrm{Var}(\tilde\xi_2) \end{pmatrix}
   \begin{pmatrix} a \\ b \end{pmatrix}$
\ --- $\mathrm{Var}(a\tilde\xi_1 + b\tilde\xi_2)$ again
\vspace{4pt}

\quad Example 3: \quad
$\mu = \begin{pmatrix} 0.05 \\ 0.05 \end{pmatrix}$, \qquad
$\Sigma = \begin{pmatrix} 0.0225 & 0.0045 \\ 0.0045 & 0.0225 \end{pmatrix}$
\vspace{14pt}
\end{board}
%% ------------------------------------------------------------------ 9
\beat{8}{The bridge to optimization}
\begin{board}
everything so far was about $\tilde\xi$ alone. now put the decision back in.
\vspace{10pt}

\quad a cost depends on both: \ $f(x, \tilde\xi)$ --- \ $x$ the \textbf{decision} we choose,
      $\tilde\xi$ the \textbf{data} we do not
\vspace{4pt}

\quad fix $x$. \ then $f(x, \tilde\xi)$ is a \textbf{random variable} --- it inherits $\tilde\xi$'s randomness
\vspace{4pt}

\quad so $\mathbb{E}\big[f(x, \tilde\xi)\big]$ is a \textbf{number}, \ one per $x$ --- and numbers can be minimized:
\[
\min_{x \in X} \;\; \mathbb{E}\big[f(x, \tilde\xi)\big]
\]

\quad {\color{punchink}that $\mathbb{E}$ is the whole design space of this course.} \ expectation is one way to turn a
      random cost into a number; CVaR, VaR and a chance constraint are others. the model above changes
      in that one slot and nowhere else.
\end{board}
\takeaway{every paradigm in this course differs in one place only: \emph{how} it collapses the randomness to a number.}
%% ------------------------------------------------------------------ 10
\beat{9}{Running example: the Markowitz portfolio}
\begin{board}
\begin{board}
a budget of $1$, \ $n$ assets. \quad decide the split today; the returns are revealed tomorrow.
\vspace{6pt}

\bul \textbf{decision:} \ $x \in \mathbb{R}^n$, \ $x_j$ = share of the budget in asset $j$
\vspace{3pt}

\bul \textbf{uncertain data:} \ $\tilde\xi = (\tilde\xi_1, \dots, \tilde\xi_n) \in \mathbb{R}^n$, \
      $\tilde\xi_j$ = return of asset $j$ \ --- a random vector with mean $\mu$, covariance matrix $\Sigma$
\vspace{3pt}

\bul \textbf{feasible set:} \ $X = \{\, x \in \mathbb{R}^n \;:\; \mathbf{1}^{\top}x = 1, \ x \ge 0 \,\}$
      \qquad ($\mathbf{1}$ = all-ones vector, \ $\mathbf{1}^{\top}x = x_1 + \cdots + x_n$)
\vspace{10pt}

\textbf{Portfolio return} \quad
$\tilde\xi^{\top}x = x_1\tilde\xi_1 + \cdots + x_n\tilde\xi_n$ \quad --- a random variable
\vspace{3pt}

\quad as a function: \ fix $x$, \quad $f(\xi) = \xi^{\top}x$, \quad $f : \mathbb{R}^n \to \mathbb{R}$
\quad --- the week 2 note's inner product, fed a random input
\vspace{3pt}

\quad loss: \ $f(x, \tilde\xi) = -\tilde\xi^{\top}x$ \qquad (a gain is a negative loss)
\vspace{6pt}

\quad Section 7's linear combinations with $c = x$: \qquad
$\mathbb{E}[\tilde\xi^{\top}x] = \mu^{\top}x$, \qquad $\mathrm{Var}(\tilde\xi^{\top}x) = x^{\top}\Sigma x$
\vspace{10pt}

\textbf{Check on Example 3} \quad $(\tilde\xi_1, \tilde\xi_2)$ = Example 3's $(\tilde\xi_1, \tilde\xi_2)$, \ half in each: \ $x = (\tfrac12, \tfrac12)$
\vspace{4pt}

\begin{center}
\begin{tabular}{@{}cc|c|c@{}}
 $\xi_1$ & $\xi_2$ & $\tfrac12\xi_1 + \tfrac12\xi_2$ & probability \\ \hline
 $-0.10$ & $-0.10$ & $-0.10$ & $0.3$ \\
 $-0.10$ & $\phantom{-}0.20$ & $\phantom{-}0.05$ & $0.2$ \\
 $\phantom{-}0.20$ & $-0.10$ & $\phantom{-}0.05$ & $0.2$ \\
 $\phantom{-}0.20$ & $\phantom{-}0.20$ & $\phantom{-}0.20$ & $0.3$ \\
\end{tabular}
\qquad\quad $\Rightarrow$ \quad PMF of $\tilde\xi^{\top}x$: \quad
\begin{tabular}{@{}l|ccc@{}}
 value       & $-0.10$ & $0.05$ & $0.20$ \\ \hline
 probability & $0.3$   & $0.4$  & $0.3$  \\
\end{tabular}
\end{center}
\vspace{4pt}

\[
\begin{aligned}
\text{from the PMF:}\quad
 & \mathbb{E}[\tilde\xi^{\top}x] = 0.3\,(-0.10) + 0.4\,(0.05) + 0.3\,(0.20) = 0.05 \\
 & \mathrm{Var}(\tilde\xi^{\top}x) = 0.3\,(-0.15)^2 + 0.4\,(0)^2 + 0.3\,(0.15)^2 = 0.0135 \\[5pt]
\text{from $\mu$ and $\Sigma$:}\quad
 & \mu^{\top}x = \tfrac12\,(0.05) + \tfrac12\,(0.05) = 0.05 \ \checkmark \\
 & x^{\top}\Sigma x = \tfrac14\,(0.0225) + \tfrac14\,(0.0225) + 2 \cdot \tfrac14\,(0.0045) = 0.0135 \ \checkmark
\end{aligned}
\]
\vspace{2pt}

  \node[align=center] at (0.5,-0.75) {asset 1 alone, \ $x = (1, 0)$\\ $\sigma = 0.15$};
\end{tikzpicture}

\quad same mean, less spread: \textbf{diversification}. \ it comes from $\rho < 1$ --- with $\rho = 1$, \ $x^{\top}\Sigma x$ would stay $0.0225$.
\vspace{12pt}

\textbf{The model} \quad (Markowitz, 1952) \quad --- least risk, subject to an expected-return target $r$:
\[
\begin{aligned}
  \min_x\quad & x^{\top}\Sigma x && \text{variance of the return (risk)}\\
  \text{s.t.}\quad & \mu^{\top}x \ge r && \text{expected return meets the target}\\
                   & \mathbf{1}^{\top}x = 1, \quad x \ge 0 && x \in X
\end{aligned}
\]

\quad the week 2 note's QP (its Section 5) with $Q = 2\Sigma$, $c = 0$; \ convex because $\Sigma \succeq 0$ (its Section 6).
\vspace{3pt}

\quad in the language of the bridge: the loss $-\tilde\xi^{\top}x$ is random; Markowitz collapses it with two numbers ---
\vspace{1pt}

\qquad its variance (objective) and its mean (constraint).
\vspace{12pt}

\textbf{Data --- the coding session} \quad $n = 3$, \ $r = 0.10$
\vspace{5pt}

\begin{center}
\begin{tabular}{@{}l|c|c|c@{}}
 asset $j$ & expected return $\mu_j$ & variance $\Sigma_{jj}$ & $\sigma_j = \sqrt{\Sigma_{jj}}$ \\ \hline
 bond      & $0.06$ & $0.010$ & $0.10$ \\
 tech      & $0.10$ & $0.040$ & $0.20$ \\
 energy    & $0.15$ & $0.090$ & $0.30$ \\
\end{tabular}
\end{center}
\vspace{4pt}

\[
x = \begin{pmatrix} x_{\text{bond}} \\ x_{\text{tech}} \\ x_{\text{energy}} \end{pmatrix},
\qquad
\mu = \begin{pmatrix} 0.06 \\ 0.10 \\ 0.15 \end{pmatrix},
\qquad
\Sigma = \begin{pmatrix} 0.010 & 0.001 & 0 \\ 0.001 & 0.040 & 0.015 \\ 0 & 0.015 & 0.090 \end{pmatrix}
\]

\quad covariances: \ bond--tech $0.001$, \ tech--energy $0.015$, \ bond--energy $0$
\vspace{6pt}

\quad written out term by term --- exactly the lines typed in the code:
\[
\begin{aligned}
  \min\quad & 0.010\,x_{\text{bond}}^2 + 0.040\,x_{\text{tech}}^2 + 0.090\,x_{\text{energy}}^2
              + 2(0.001)\,x_{\text{bond}}x_{\text{tech}} + 2(0.015)\,x_{\text{tech}}x_{\text{energy}} \\
  \text{s.t.}\quad & 0.06\,x_{\text{bond}} + 0.10\,x_{\text{tech}} + 0.15\,x_{\text{energy}} \ge 0.10 \\
                   & x_{\text{bond}} + x_{\text{tech}} + x_{\text{energy}} \le 1 \\
                   & x_{\text{bond}},\ x_{\text{tech}},\ x_{\text{energy}} \ge 0
\end{aligned}
\]

\quad budget ``$\le 1$'': leftover sits in cash earning $0$ --- never helps, so the sum is $1$ at the optimum
\vspace{6pt}

\textbf{Solution}
\vspace{3pt}

\quad $\boxed{x^{\star} \approx (0.39,\ 0.29,\ 0.31), \qquad x^{\star\top}\Sigma x^{\star} \approx 0.0169, \qquad \sigma \approx 0.130}$
\qquad $\mu^{\top}x^{\star} = 0.10$ --- target met exactly
\vspace{4pt}

\quad all in tech: \ same expected return $0.10$, \ but $\sigma = 0.20$. \quad the mix cuts $\sigma$ by about a third.
\vspace{3pt}

\quad every asset gets a share --- not a vertex (week 2 note, Section 5).
\vspace{6pt}
